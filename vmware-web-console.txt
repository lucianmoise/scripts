// Functia de mai jos se foloseste intr-o console de js in pagina de web pentru a face paste la un text atunci cand nu functioneaza Paste cu mouse sau altceva
// am folosit-o in HF3 pentru VMware web client
// fullMap array este pentru a da paste cand masina remote are tastatura in FR

async function typeUniversalAZERTY(text, delay = 30) {
    const wmksContainer = document.querySelector('.wmks, #wmksContainer, canvas');
    if (!wmksContainer) {
        console.error("Nu s-a găsit elementul consolei VMware!");
        return;
    }
    wmksContainer.focus();

    // Mapare completă: Caracterul dorit -> { tasta fizică, keyCode, are Nevoie de Shift? }
    const fullMap = {
        // Litere mici
        'a': { code: 'KeyQ', keyCode: 81, shift: false },
        'b': { code: 'KeyB', keyCode: 66, shift: false },
        'c': { code: 'KeyC', keyCode: 67, shift: false },
        'd': { code: 'KeyD', keyCode: 68, shift: false },
        'e': { code: 'KeyE', keyCode: 69, shift: false },
        'f': { code: 'KeyF', keyCode: 70, shift: false },
        'g': { code: 'KeyG', keyCode: 71, shift: false },
        'h': { code: 'KeyH', keyCode: 72, shift: false },
        'i': { code: 'KeyI', keyCode: 73, shift: false },
        'j': { code: 'KeyJ', keyCode: 74, shift: false },
        'k': { code: 'KeyK', keyCode: 75, shift: false },
        'l': { code: 'KeyL', keyCode: 76, shift: false },
        'm': { code: 'Semicolon', keyCode: 186, shift: false },
        'n': { code: 'KeyN', keyCode: 78, shift: false },
        'o': { code: 'KeyO', keyCode: 79, shift: false },
        'p': { code: 'KeyP', keyCode: 80, shift: false },
        'q': { code: 'KeyA', keyCode: 65, shift: false },
        'r': { code: 'KeyR', keyCode: 82, shift: false },
        's': { code: 'KeyS', keyCode: 83, shift: false },
        't': { code: 'KeyT', keyCode: 84, shift: false },
        'u': { code: 'KeyU', keyCode: 85, shift: false },
        'v': { code: 'KeyV', keyCode: 86, shift: false },
        'w': { code: 'KeyZ', keyCode: 90, shift: false },
        'x': { code: 'KeyX', keyCode: 88, shift: false },
        'y': { code: 'KeyY', keyCode: 89, shift: false },
        'z': { code: 'KeyW', keyCode: 87, shift: false },

        // Litere mari (Folosesc aceleași taste, dar cu Shift activat)
        'A': { code: 'KeyQ', keyCode: 81, shift: true },
        'B': { code: 'KeyB', keyCode: 66, shift: true },
        'C': { code: 'KeyC', keyCode: 67, shift: true },
        'D': { code: 'KeyD', keyCode: 68, shift: true },
        'E': { code: 'KeyE', keyCode: 69, shift: true },
        'F': { code: 'KeyF', keyCode: 70, shift: true },
        'G': { code: 'KeyG', keyCode: 71, shift: true },
        'H': { code: 'KeyH', keyCode: 72, shift: true },
        'I': { code: 'KeyI', keyCode: 73, shift: true },
        'J': { code: 'KeyJ', keyCode: 74, shift: true },
        'K': { code: 'KeyK', keyCode: 75, shift: true },
        'L': { code: 'KeyL', keyCode: 76, shift: true },
        'M': { code: 'Semicolon', keyCode: 186, shift: true },
        'N': { code: 'KeyN', keyCode: 78, shift: true },
        'O': { code: 'KeyO', keyCode: 79, shift: true },
        'P': { code: 'KeyP', keyCode: 80, shift: true },
        'Q': { code: 'KeyA', keyCode: 65, shift: true },
        'R': { code: 'KeyR', keyCode: 82, shift: true },
        'S': { code: 'KeyS', keyCode: 83, shift: true },
        'T': { code: 'KeyT', keyCode: 84, shift: true },
        'U': { code: 'KeyU', keyCode: 85, shift: true },
        'V': { code: 'KeyV', keyCode: 86, shift: true },
        'W': { code: 'KeyZ', keyCode: 90, shift: true },
        'X': { code: 'KeyX', keyCode: 88, shift: true },
        'Y': { code: 'KeyY', keyCode: 89, shift: true },
        'Z': { code: 'KeyW', keyCode: 87, shift: true },

        // Cifre (În AZERTY cifrele se tastează obligatoriu cu SHIFT!)
        '1': { code: 'Digit1', keyCode: 49, shift: true },
        '2': { code: 'Digit2', keyCode: 50, shift: true },
        '3': { code: 'Digit3', keyCode: 51, shift: true },
        '4': { code: 'Digit4', keyCode: 52, shift: true },
        '5': { code: 'Digit5', keyCode: 53, shift: true },
        '6': { code: 'Digit6', keyCode: 54, shift: true },
        '7': { code: 'Digit7', keyCode: 55, shift: true },
        '8': { code: 'Digit8', keyCode: 56, shift: true },
        '9': { code: 'Digit9', keyCode: 57, shift: true },
        '0': { code: 'Digit0', keyCode: 48, shift: true },

        // Caractere speciale uzuale în Linux
        ' ': { code: 'Space', keyCode: 32, shift: false },
        '\n': { code: 'Enter', keyCode: 13, shift: false },
        '-': { code: 'Digit6', keyCode: 54, shift: false },     
        '_': { code: 'Digit8', keyCode: 56, shift: false },     
        '/': { code: 'Period', keyCode: 190, shift: true },     
        ':': { code: 'Period', keyCode: 190, shift: false },    
        '!': { code: 'Slash', keyCode: 191, shift: false },     
        '=': { code: 'Equal', keyCode: 187, shift: false },     
        '+': { code: 'Equal', keyCode: 187, shift: true },      
        '*': { code: 'Backslash', keyCode: 220, shift: false }, 
        ';': { code: 'M', keyCode: 77, shift: false }
    };

    function sendKeyEvent(type, targetCode, targetKeyCode, isShiftPressed) {
        const ev = new KeyboardEvent(type, {
            key: '', 
            code: targetCode,
            keyCode: targetKeyCode,
            which: targetKeyCode,
            shiftKey: isShiftPressed,
            bubbles: true,
            cancelable: true
        });
        wmksContainer.dispatchEvent(ev);
    }

    for (let char of text) {
        if (fullMap[char]) {
            const keyConfig = fullMap[char];

            // 1. Apăsăm fizic Shift-ul stânga dacă scriptul o cere și lăsăm o mică pauză ca VMware să proceseze starea
            if (keyConfig.shift) {
                sendKeyEvent('keydown', 'ShiftLeft', 16, true);
                await new Promise(r => setTimeout(r, 15)); 
            }

            // 2. Apăsăm și eliberăm tasta principală
            sendKeyEvent('keydown', keyConfig.code, keyConfig.keyCode, keyConfig.shift);
            sendKeyEvent('keypress', keyConfig.code, keyConfig.keyCode, keyConfig.shift);
            
            await new Promise(r => setTimeout(r, delay));
            
            sendKeyEvent('keyup', keyConfig.code, keyConfig.keyCode, keyConfig.shift);

            // 3. Eliberăm fizic tasta Shift stânga
            if (keyConfig.shift) {
                await new Promise(r => setTimeout(r, 15));
                sendKeyEvent('keyup', 'ShiftLeft', 16, false);
            }
            
            // O scurtă pauză între caractere diferite ca să nu se suprapună pachetele WebSockets
            await new Promise(r => setTimeout(r, 15));
        } else {
            let kc = char.charCodeAt(0);
            sendKeyEvent('keydown', '', kc, false);
            await new Promise(r => setTimeout(r, delay));
            sendKeyEvent('keyup', '', kc, false);
        }
    }
    console.log("Comandă trimisă cu Shift asincron pe AZERTY!");
}

// TESTEAZĂ TEXTUL MAI JOS (Verifică dacă cifrele sau majusculele apar acum corect)
typeUniversalAZERTY("TextToPaste");
